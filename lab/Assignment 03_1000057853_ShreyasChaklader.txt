import java.util.Scanner;
public class assignment03_task2{
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
        System.out.print("Please Enter number of Herbs (H): ");
        int h = sc.nextInt();
        System.out.print("Please Enter number of Crystals (C): ");
        int c = sc.nextInt();
        int potion = 0;
        while (h >= 3 && c >= 2){
            h -= 3;
            c -= 2;
            potion++;
            System.out.println("Potion-" + potion + " created");
            System.out.println("Remaining Herbs: " + h + " Remaining Crystals: " + c);
        }
        System.out.println("Potions Created: " + potion);
        if(potion % 2 == 0){
            System.out.println("Stable Brew");
        } 
        else{
            System.out.println("Volatile Elixir");
        }
    }
}

        System.out.println("Enter the N-digit vault code: ");
        int code = sc.nextInt();
        int n = 0;
        int temp = code;
        while(temp > 0){
            n++;
            temp = temp / 10;
        }
        int divider = 1;
        while (n > 1){
            divider *= 10;
            n--;
        }
        while (divider > 0){
            int digit = code / divider;
            System.out.print(digit);
            code %= divider;
            divider /= 10;
            if (divider > 0) {
                System.out.print(", ");
            }
        }
    }
}

import java.util.Scanner;
public class assignment03_task6{
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
        int num = sc.nextInt();
        int divisors = 0;
        for(int i = 1; i <= num; i++){
            if(num % i == 0){
                divisors += 1;
            }
        }
        if(divisors == 2){
            System.out.println(num + " is a prime number");
        }
        else{
            System.out.println(num + " is not a prime number");
        }
        int Sum = 0;
        for(int i = 1; i < num; i++){
            if(num % i == 0){
                Sum += i;
            }
        }
        if(Sum == num){
            System.out.println(num + " is a perfect number");
        } 
        else{
            System.out.println(num + " is not a perfect number");
        }
    }
}
